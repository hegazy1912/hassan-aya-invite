var ADMIN_KEY = '__ADMIN_KEY__';
var SHEET_NAME = 'الضيوف';
var HEADERS = ['الكود', 'الاسم', 'الموبايل', 'الحالة', 'وقت الرد', 'وقت الإرسال'];
var STATUS = { yes: 'جاي', no: 'مش جاي', pending: 'لسه' };

function doGet(e) {
  var p = (e && e.parameter) || {};
  var cb = safeCallback_(p.callback);
  var out;
  var lock = LockService.getScriptLock();
  var locked = false;
  try {
    locked = lock.waitLock(15000);
    var sh = getSheet_();
    var action = String(p.action || 'rsvp').toLowerCase();
    if (action === 'rsvp') {
      out = doRsvp_(sh, p);
    } else {
      if (String(p.key || '') !== ADMIN_KEY) throw new Error('unauthorized');
      if (action === 'list') out = doList_(sh);
      else if (action === 'add') out = doAdd_(sh, p);
      else if (action === 'addbulk') out = doAddBulk_(sh, p);
      else if (action === 'sent') out = doSent_(sh, p);
      else if (action === 'status') out = doStatus_(sh, p);
      else if (action === 'del') out = doDelete_(sh, p);
      else throw new Error('unknown_action');
    }
    out.ok = true;
  } catch (err) {
    out = { ok: false, error: String((err && err.message) || err) };
  } finally {
    if (locked) {
      try { lock.releaseLock(); } catch (ignore) {}
    }
  }
  return reply_(out, cb);
}

function getSheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) throw new Error('open_this_script_from_the_sheet');
  var sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) sh = ss.insertSheet(SHEET_NAME);
  if (String(sh.getRange(1, 1).getValue()) !== HEADERS[0]) {
    if (sh.getLastRow() > 0) sh = ss.insertSheet(SHEET_NAME + ' ' + new Date().getTime());
    sh.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]);
    sh.setFrozenRows(1);
    sh.getRange(1, 1, 1, HEADERS.length).setFontWeight('bold').setBackground('#f3e9d8');
    sh.setColumnWidth(1, 150);
    sh.setColumnWidth(2, 220);
    sh.setColumnWidth(3, 150);
    sh.setColumnWidth(4, 120);
    sh.setColumnWidth(5, 165);
    sh.setColumnWidth(6, 165);
    var dv = SpreadsheetApp.newDataValidation()
      .requireValueInList([STATUS.yes, STATUS.no, STATUS.pending], true)
      .setAllowInvalid(true)
      .build();
    sh.getRange(2, 4, 2000, 1).setDataValidation(dv);
  }
  return sh;
}

function findRow_(sh, id) {
  id = String(id || '').trim();
  if (!id) return 0;
  var last = sh.getLastRow();
  if (last < 2) return 0;
  var vals = sh.getRange(2, 1, last - 1, 1).getValues();
  for (var i = 0; i < vals.length; i++) {
    if (String(vals[i][0]) === id) return i + 2;
  }
  return 0;
}

function makeId_() {
  return Utilities.getUuid().replace(/-/g, '').slice(0, 10);
}

function normPhone_(v) {
  var s = String(v || '').replace(/[^\d]/g, '');
  if (s.indexOf('00') === 0) s = s.slice(2);
  if (s.length === 11 && s.charAt(0) === '0') s = '20' + s.slice(1);
  else if (s.length === 10 && s.charAt(0) === '1') s = '20' + s;
  return s;
}

function iso_(v) {
  if (!v) return '';
  var d = (v instanceof Date) ? v : new Date(v);
  return isNaN(d.getTime()) ? '' : d.toISOString();
}

function doRsvp_(sh, p) {
  var id = String(p.id || '').trim();
  var name = String(p.name || '').trim().slice(0, 120);
  var r = String(p.r || 'yes').toLowerCase();
  var status = (r === 'no' || r === '0' || r === 'declined') ? STATUS.no : STATUS.yes;
  if (!id) throw new Error('missing_id');
  var row = findRow_(sh, id);
  if (!row) throw new Error('unknown_guest');
  var now = new Date();
  sh.getRange(row, 4).setValue(status);
  sh.getRange(row, 5).setValue(now);
  if (name && !String(sh.getRange(row, 2).getValue())) sh.getRange(row, 2).setValue(name);
  return { id: id, name: name, status: status };
}

function doList_(sh) {
  var last = sh.getLastRow();
  if (last < 2) return { guests: [], count: 0 };
  var vals = sh.getRange(2, 1, last - 1, HEADERS.length).getValues();
  var out = [];
  for (var i = 0; i < vals.length; i++) {
    var r = vals[i];
    if (String(r[0]).trim() === '') continue;
    out.push({
      id: String(r[0]),
      name: String(r[1] || ''),
      phone: String(r[2] || ''),
      status: String(r[3] || '').trim() || STATUS.pending,
      respondedAt: iso_(r[4]),
      sentAt: iso_(r[5])
    });
  }
  return { guests: out, count: out.length };
}

function doAdd_(sh, p) {
  var name = String(p.name || '').trim().slice(0, 120);
  if (!name) throw new Error('name_required');
  var id = makeId_();
  sh.appendRow([id, name, normPhone_(p.phone || ''), STATUS.pending, '', '']);
  return { id: id, name: name };
}

function doAddBulk_(sh, p) {
  var arr;
  try {
    arr = JSON.parse(String(p.data || '[]'));
  } catch (err) {
    throw new Error('bad_data');
  }
  if (!Array.isArray(arr)) throw new Error('bad_data');
  var rows = [];
  for (var i = 0; i < arr.length; i++) {
    var g = arr[i] || {};
    var name = String(g.name || '').trim().slice(0, 120);
    if (!name) continue;
    rows.push([makeId_(), name, normPhone_(g.phone || ''), STATUS.pending, '', '']);
  }
  if (!rows.length) return { added: 0 };
  sh.getRange(sh.getLastRow() + 1, 1, rows.length, HEADERS.length).setValues(rows);
  return { added: rows.length };
}

function doSent_(sh, p) {
  var id = String(p.id || '').trim();
  var row = findRow_(sh, id);
  if (!row) throw new Error('not_found');
  var v = String(p.v == null ? '1' : p.v);
  sh.getRange(row, 6).setValue(v === '0' ? '' : new Date());
  return { id: id, sent: v !== '0' };
}

function doStatus_(sh, p) {
  var id = String(p.id || '').trim();
  var status = String(p.status || '').trim();
  if ([STATUS.yes, STATUS.no, STATUS.pending].indexOf(status) < 0) throw new Error('bad_status');
  var row = findRow_(sh, id);
  if (!row) throw new Error('not_found');
  sh.getRange(row, 4).setValue(status);
  if (status !== STATUS.pending) sh.getRange(row, 5).setValue(new Date());
  return { id: id, status: status };
}

function doDelete_(sh, p) {
  var id = String(p.id || '').trim();
  var row = findRow_(sh, id);
  if (!row) throw new Error('not_found');
  if (row < 2) throw new Error('bad_row');
  sh.deleteRow(row);
  return { id: id, deleted: true };
}

function safeCallback_(cb) {
  cb = String(cb || '');
  return /^[A-Za-z_$][0-9A-Za-z_$]*(\.[A-Za-z_$][0-9A-Za-z_$]*)*$/.test(cb) ? cb : '';
}

function reply_(obj, cb) {
  var json = JSON.stringify(obj);
  json = json
    .replace(/</g, '\\u003c')
    .replace(/>/g, '\\u003e')
    .replace(/&/g, '\\u0026')
    .replace(/\u2028/g, '\\u2028')
    .replace(/\u2029/g, '\\u2029');
  if (cb) {
    return ContentService.createTextOutput(cb + '(' + json + ');')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON);
}
